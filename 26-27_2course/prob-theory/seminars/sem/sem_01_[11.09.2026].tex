\subsection{Задача 1}

\begin{solution}
    \textbf{Модель: классическая.}
    Карточки нумеруем: А --- $1,2,3$; Е --- $4$; И --- $5$;
    К --- $6$; М --- $7,8$; Т --- $9,10$.
    \[
        \omega=(i_1,\ldots,i_{10}),\qquad
        i_j\in\{1,\ldots,10\},\quad i_j\neq i_k\ (j\neq k).
    \]
    \[
        |\Omega|=10!,\qquad
        \mathcal{A}=\{\text{получилось слово «математика»}\},
        \qquad |\mathcal{A}|=3!\,2!\,2!=24.
    \]
    \[
        \Pr(\mathcal{A})=\frac{|\mathcal{A}|}{|\Omega|}
        =\boxed{\frac{24}{10!}}.
    \]
\end{solution}

\subsection{Задача 2}

\begin{solution}
    \textbf{Модель: классическая.}
    $i_j$ --- номер выбранной клетки в $j$-й строке.
    \[
        \omega=(i_1,\ldots,i_5),\qquad
        \Omega=\{1,\ldots,5\}^5,\qquad |\Omega|=5^5.
    \]
    \[
        \mathcal{A}=\{\text{получилось слово «ДЖИНС»}\},\qquad
        |\mathcal{A}|=1,\qquad
        \Pr(\mathcal{A})=\boxed{\frac1{5^5}}.
    \]
\end{solution}

\subsection{Задача 3}

\begin{solution}
    \textbf{Модель: классическая, упорядоченный выбор.}
    \[
        \Omega=\{(i_1,i_2):i_1,i_2\in\{1,\ldots,n\},\ i_1\neq i_2\},
        \qquad |\Omega|=n(n-1)=A_n^2.
    \]
    \[
        \mathcal{A}=\{i_1<k<i_2\}\sqcup\{i_2<k<i_1\},
        \qquad |\mathcal{A}|=2(k-1)(n-k).
    \]
    \[
        \Pr(\mathcal{A})=\boxed{\frac{2(k-1)(n-k)}{n(n-1)}}.
    \]
    Для неупорядоченного выбора:
    \[
        \Pr(\mathcal{A})=\frac{(k-1)(n-k)}{C_n^2}.
    \]
\end{solution}

\subsection{Задача 4}

\begin{solution}
    \textbf{Модель: классическая.}
    Нумеруем места за столом; $\omega=(i_1,\ldots,i_n)$ ---
    перестановка друзей, $|\Omega|=n!$.

    \textbf{а)} $\mathcal{A}$ --- B непосредственно слева от A.
    \[
        |\mathcal{A}|=n\cdot1\cdot(n-2)!,
        \qquad
        \Pr(\mathcal{A})=\frac{n(n-2)!}{n!}
        =\boxed{\frac1{n-1}}.
    \]
    При фиксированном месте A: $n-1$ равновероятных мест для B,
    подходящее одно.

    \textbf{б)} $\mathcal{B}$ --- A непосредственно справа от B,
    C непосредственно слева от B.
    \[
        |\mathcal{B}|=n\cdot1\cdot1\cdot(n-3)!,
        \qquad
        \Pr(\mathcal{B})=\frac{n(n-3)!}{n!}
        =\boxed{\frac1{(n-1)(n-2)}}.
    \]
\end{solution}

\subsection{Задача 5}

\begin{solution}
    \textbf{Модель: классическая, неупорядоченная выборка без возвращения.}
    \[
        \Omega=\bigl\{\{i_1,i_2,i_3,i_4\}\subseteq\{1,\ldots,16\}:
        |\{i_1,i_2,i_3,i_4\}|=4\bigr\},
        \qquad |\Omega|=C_{16}^4.
    \]
    \textbf{а)} $\mathcal{A}$ --- ровно два чёрных шара.
    \[
        |\mathcal{A}|=C_6^2C_{10}^2,\qquad
        \Pr(\mathcal{A})=\boxed{\frac{C_6^2C_{10}^2}{C_{16}^4}}.
    \]
    \textbf{б)} $\mathcal{B}$ --- хотя бы один чёрный шар.
    \[
        |\overline{\mathcal{B}}|=C_{10}^4,\qquad
        \Pr(\mathcal{B})=1-\Pr(\overline{\mathcal{B}})
        =\boxed{1-\frac{C_{10}^4}{C_{16}^4}}.
    \]
\end{solution}

\subsection{Задача 6}

\begin{solution}
    \textbf{Модель: классическая, неупорядоченная выборка без возвращения.}
    \[
        \omega=\{i_1,\ldots,i_6\}\subseteq\{1,\ldots,52\},
        \qquad |\omega|=6,\qquad |\Omega|=C_{52}^6.
    \]
    \textbf{а)} $\mathcal{A}$ --- выбран король пик.
    \[
        |\mathcal{A}|=C_{51}^5,\qquad
        \Pr(\mathcal{A})=\boxed{\frac{C_{51}^5}{C_{52}^6}}.
    \]

    \textbf{б)} $\mathcal{B}$ --- представлены все четыре масти.
    Распределения числа карт по мастям: $(3,1,1,1)$ или $(2,2,1,1)$.
    \[
        \Pr(\mathcal{B})
        =\boxed{\frac{4C_{13}^3(C_{13}^1)^3+
            C_4^2(C_{13}^2)^2(C_{13}^1)^2}{C_{52}^6}}.
    \]
\end{solution}
