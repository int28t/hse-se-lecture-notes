\newcounter{questiontopic}[subsection]
\newcommand{\registertopic}[3]{%
  \ifcsdef{topics.#1}{\csgappto{topics.#1}{\enspace\textperiodcentered\enspace}}{}%
  \csgappto{topics.#1}{\hyperlink{#2}{#3}}}
\makeatletter
\newcommand{\topicmark}[1]{%
  \stepcounter{questiontopic}%
  \edef\topicdest{topic.\arabic{subsection}.\arabic{questiontopic}}%
  \hypertarget{\topicdest}{}%
  \bookmark[level=3,dest=\topicdest]{#1}%
  \addtocontents{toc}{\protect\contentsline{subsubsection}%
    {\protect\numberline{\arabic{subsection}.\arabic{questiontopic}}#1}%
    {\thepage}{\topicdest}\protected@file@percent}%
  \protected@write\@auxout{}{\string\registertopic{\arabic{subsection}}{\topicdest}{#1}}%
  \ignorespaces}
\makeatother
